// Task 2: Advanced Querying


// a) Books rated above 4 ($gt)
db.book_metadata.find(
  { avgRating: { $gt: 4 } },
  { title: 1, avgRating: 1, _id: 0 }
);

// b) Books available in specific formats ($in)
db.book_metadata.find(
  { formats: { $in: ["Digital", "Audiobook"] } },
  { title: 1, formats: 1, _id: 0 }
);

// c) Combine both with $and (implicit)
db.book_metadata.find({
  avgRating: { $gt: 4 },
  formats: { $in: ["Digital"] }
});

// d) Regex query - books with "Digital" format
db.book_metadata.find(
  { formats: { $regex: "Digital", $options: "i" } },
  { title: 1, formats: 1 }
);

// e) Regex query - reviews mentioning "Digital" in comments (nested field)
db.book_metadata.find(
  { "reviews.comment": { $regex: "Digital", $options: "i" } },
  { title: 1, "reviews.comment": 1 }
);



// Task 3: Aggregation Pipeline - "Top Rated" Report


db.book_metadata.aggregate([
  { $match: { publishedYear: { $gt: 2020 } } },
  { $unwind: "$reviews" },
  {
    $group: {
      _id: "$title",
      avgReviewRating: { $avg: "$reviews.rating" },
      totalReviews: { $sum: 1 },
      author: { $first: "$author" }
    }
  },
  { $sort: { avgReviewRating: -1 } },
  {
    $project: {
      _id: 0,
      title: "$_id",
      author: 1,
      avgReviewRating: { $round: ["$avgReviewRating", 2] },
      totalReviews: 1
    }
  }
]);


// Task 4: Indexing for Speed


// a) Create a text index on review comments
db.book_metadata.createIndex({ "reviews.comment": "text" });

// b) Query using the text index
db.book_metadata.find(
  { $text: { $search: "Digital" } },
  { title: 1, score: { $meta: "textScore" } }
).sort({ score: { $meta: "textScore" } });

// c) Verify index usage with .explain()
db.book_metadata.find(
  { $text: { $search: "Digital" } }
).explain("executionStats");

// d) Compare against a non-indexed regex scan
db.book_metadata.find(
  { "reviews.comment": { $regex: "Digital", $options: "i" } }
).explain("executionStats");

// e) Supporting index for common queries (multikey & single field indexes)
db.book_metadata.createIndex({ formats: 1 });
db.book_metadata.createIndex({ avgRating: -1 });

// Explain plan verifying usage of avgRating index
db.book_metadata.find({ avgRating: { $gt: 4 } }).explain("executionStats");
